║                                                                          ║

🚨 PROBLEM: Local files would be overwritten by git pull

✅ SOLUTION: Backup local files, then pull fresh from GitHub

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

📋 RUN THESE COMMANDS ON VPS:

# 1. Backup existing files
cd /root/doh
mkdir -p backup-$(date +%s)
mv add-https.sh fix-https.sh fix-upstream-doh.sh optimize-for-xbox.sh setup-letsencrypt.sh verify-ipv6-ready.sh backup-*/

# 2. Now pull from GitHub
git pull origin main

# 3. Make new scripts executable
chmod +x *.sh

# 4. Check what you got
ls -lh *.sh

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

OR USE GIT STASH (Alternative):

# 1. Stash local changes
cd /root/doh
git stash

# 2. Pull from GitHub
git pull origin main

# 3. Make scripts executable
chmod +x *.sh

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

AFTER PULLING, RUN SMART DNS SETUP:

./setup-xbox-proxy.sh
./integrate-coredns-smartdns.sh
./test-smartdns.sh
